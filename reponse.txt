13: select
    c.id_categorie,
    c.libelle,
    round(avg(em.date_retour - em.date_emprunt)::numeric, 1) as duree_moyenne_emprunt
    from categorie c
    join livre l
    on c.id_categorie = l.id_categorie
    join exemplaire ex
    on l.id_livre = ex.id_livre
    join emprunt em
    on ex.id_exemplaire = em.id_exemplaire
    where em.date_retour is not NULL
    group by c.id_categorie;

14:select
   a.id_adherent,
   a.nom,
   count(em.id_emprunt) as active_loan_count
   from adherent a
   join emprunt em
   on a.id_adherent = em.id_adherent
   where em.date_retour is null
   group by
   a.id_adherent,
   a.nom
   having count(em.id_emprunt) > 1;

15: select
    c.id_categorie,
    c.libelle,
    count(distinct l.id_livre) as book_count,
    count(distinct ex.id_exemplaire) as copy_count,
    count(em.id_emprunt) as total_loan_count
    from categorie c
    left join livre l
    on c.id_categorie = l.id_categorie
    left join exemplaire ex
    on l.id_livre = ex.id_livre
    left join emprunt em
    on ex.id_exemplaire = em.id_exemplaire
    group by
    c.id_categorie,
    c.libelle;
